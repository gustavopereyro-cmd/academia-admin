<script>
const API_URL = 'https://script.google.com/macros/s/AKfycbzyGiSnozdfWIIVSy1PY25WqgGS1l8MSssblAv3diG31t_TOCeap_YO-jfeSg7TIY2Q/exec';

let DATA = { cursos: [], clases: [], materiales: [], alumnos: [], inscripciones: [], egresados: [], alumnosConCursos: [] };
let currentTab = 'cursos';

async function api(action, payload) {
  const res = await fetch(API_URL, {
    method: 'POST',
    headers: { 'Content-Type': 'text/plain;charset=utf-8' },
    body: JSON.stringify({ action, ...payload })
  });
  const json = await res.json();
  if (json.error) throw new Error(json.error);
  return json;
}

async function doLogin() {
  const usuario = document.getElementById('login-user').value.trim();
  const password = document.getElementById('login-pass').value;
  const errorEl = document.getElementById('login-error');
  errorEl.textContent = '';
  if (!usuario || !password) { errorEl.textContent = 'Completá usuario y contraseña.'; return; }
  try {
    const result = await api('login', { usuario, password });
    if (result.success) {
      sessionStorage.setItem('panel_logueado', '1');
      document.getElementById('login-screen').style.display = 'none';
      document.getElementById('app').style.display = 'block';
      loadAll();
    } else {
      errorEl.textContent = 'Usuario o contraseña incorrectos.';
    }
  } catch (e) {
    errorEl.textContent = 'No se pudo conectar con el servidor: ' + e.message;
  }
}
function logout() {
  sessionStorage.removeItem('panel_logueado');
  location.reload();
}
if (sessionStorage.getItem('panel_logueado') === '1') {
  document.getElementById('login-screen').style.display = 'none';
  document.getElementById('app').style.display = 'block';
  loadAll();
}

async function loadAll() {
  try {
    const [cursos, clases, materiales, alumnos, inscripciones, egresados, alumnosConCursos] = await Promise.all([
      api('getCursos', {}), api('getClases', {}), api('getMateriales', {}),
      api('getAlumnos', {}), api('getInscripciones', {}), api('getEgresados', {}),
      api('getAlumnosConCursos', {})
    ]);
    DATA.cursos = cursos; DATA.clases = clases; DATA.materiales = materiales;
    DATA.alumnos = alumnos; DATA.inscripciones = inscripciones;
    DATA.egresados = egresados;
    DATA.alumnosConCursos = alumnosConCursos;
    renderAll();
  } catch (e) {
    showToast('Error cargando datos: ' + e.message, true);
  }
}

function formatearPrecio(valor) {
  if (valor === '' || valor === null || valor === undefined) return '';
  const num = Number(valor);
  if (isNaN(num)) return String(valor);
  return num.toLocaleString('es-AR');
}

function cursoCerrado(curso) {
  const f = curso.Fecha_Fin_Publicacion;
  if (!f) return false;
  const fecha = new Date(f);
  if (isNaN(fecha.getTime())) return false;
  const hoy = new Date();
  hoy.setHours(0,0,0,0);
  return fecha < hoy;
}

function emptyRow(colspan, msg) { return `<tr class="empty-row"><td colspan="${colspan}">${msg}</td></tr>`; }
function esc(v) { return (v === undefined || v === null) ? '' : String(v).replace(/[&<>"]/g, s => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[s])); }
function cursoNombre(id) { const c = DATA.cursos.find(x => x.ID_Curso === id); return c ? esc(c.Nombre) : esc(id); }
function claseTitulo(id) { const c = DATA.clases.find(x => x.ID_Clase === id); return c ? esc(c.Titulo) : esc(id); }
function alumnoNombre(id) { const a = DATA.alumnos.find(x => String(x.ID_Alumno) === String(id)); return a ? esc(a.Nombre) : esc(id); }

function switchTab(tab) {
  currentTab = tab;
  document.querySelectorAll('.nav-btn').forEach(b => b.classList.toggle('active', b.dataset.tab === tab));
  ['cursos','clases','alumnos','inscripciones','certificados'].forEach(t => {
    const el = document.getElementById('tab-' + t);
    if (el) el.style.display = (t === tab) ? 'block' : 'none';
  });
}

async function cerrarCurso(row) {
  if (!confirm('¿Cerrar este curso? Dejará de aparecer en el catálogo público, pero los alumnos ya inscriptos seguirán viéndolo en el aula.')) return;
  try {
    await api('cerrarCurso', { row });
    showToast('Curso cerrado correctamente.');
    await loadAll();
  } catch (e) {
    showToast('Error al cerrar: ' + e.message, true);
  }
}

async function reabrirCurso(row) {
  if (!confirm('¿Reabrir este curso? Volverá a aparecer en el catálogo público.')) return;
  try {
    await api('reabrirCurso', { row });
    showToast('Curso reabierto correctamente.');
    await loadAll();
  } catch (e) {
    showToast('Error al reabrir: ' + e.message, true);
  }
}

async function borrarCurso(row) {
  if (!confirm('¿Borrar DEFINITIVAMENTE este curso?\n\nSe borrarán también sus clases, materiales e inscripciones. Esta acción no se puede deshacer.')) return;
  if (!confirm('¿Estás SEGURO?\n\nLos certificados ya emitidos NO se borran, pero el curso desaparecerá para siempre.')) return;
  try {
    await api('deleteCurso', { row });
    showToast('Curso borrado correctamente.');
    await loadAll();
  } catch (e) {
    showToast('Error al borrar: ' + e.message, true);
  }
}

async function removeItem(label, key, row) {
  if (!confirm(`¿Seguro que querés borrar este ${label.toLowerCase()}? Esta acción no se puede deshacer.`)) return;
  const actionMap = { cursos:'deleteCurso', clases:'deleteClase', materiales:'deleteMaterial', alumnos:'deleteAlumno', inscripciones:'deleteInscripcion', egresados:'deleteEgresado' };
  try {
    await api(actionMap[key], { row });
    showToast(label + ' borrado correctamente.');
    await loadAll();
  } catch (e) {
    showToast('Error al borrar: ' + e.message, true);
  }
}

async function emitirPendientes() {
  const btn = document.getElementById('btn-emitir');
  if (!confirm('¿Emitir todos los certificados pendientes? Esto puede tardar unos segundos.')) return;
  if (btn) { btn.disabled = true; btn.textContent = 'Emitiendo...'; }
  try {
    const res = await api('emitirPendientes', {});
    if (res.error) throw new Error(res.error);
    const msg = `Emitidos: ${res.emitidos}. Errores: ${res.errores}.`;
    if (res.errores > 0) {
      const fallados = (res.detalle || []).filter(d => !d.ok).map(d => `${d.cert} - ${d.nombre}: ${d.error}`).join('\n');
      console.warn('Errores:\n' + fallados);
    }
    showToast(msg, res.errores > 0);
    await loadAll();
  } catch (e) {
    showToast('Error al emitir: ' + e.message, true);
  } finally {
    if (btn) { btn.disabled = false; btn.textContent = 'Emitir pendientes'; }
  }
}

async function enviarPorMail(row) {
  const eg = DATA.egresados.find(x => x._row === row);
  if (!eg) return;
  if (!eg.URL_PDF) {
    showToast('Primero tenés que emitir el certificado.', true);
    return;
  }
  try {
    const res = await api('getEmailEgresado', { dni: eg.DNI });
    const email = res.email || '';
    if (!email) {
      showToast('No se encontró un email para ese DNI en la hoja Alumnos.', true);
      return;
    }
    const asunto = encodeURIComponent('Tu certificado - Academia Mariana Barrera');
    const cuerpo = encodeURIComponent(
      `Hola ${eg.Nombre_Alumno},\n\nTe enviamos el certificado correspondiente a "${eg.Curso}".\n\nPodés verlo y descargarlo acá:\n${eg.URL_PDF}\n\nCualquier consulta, quedamos a disposición.\n\nAcademia Profesional Mariana Barrera`
    );
    const gmailUrl = `https://mail.google.com/mail/?view=cm&fs=1&to=${encodeURIComponent(email)}&su=${asunto}&body=${cuerpo}`;
    window.open(gmailUrl, '_blank');
  } catch (e) {
    showToast('Error al obtener el email: ' + e.message, true);
  }
}

function hoyISO() {
  const d = new Date();
  const y = d.getFullYear();
  const m = String(d.getMonth() + 1).padStart(2, '0');
  const dd = String(d.getDate()).padStart(2, '0');
  return y + '-' + m + '-' + dd;
}

function openForm(kind, row) {
  editingRow = row || null;
  const def = FORM_DEFS[kind];
  const existing = row ? DATA[def.dataKey].find(x => x._row === row) : null;

  let html = `<h3>${existing ? 'Editar' : 'Nuevo'} ${def.title.toLowerCase()}</h3><div id="form-fields">`;
  def.fields.forEach(f => { html += renderField(f, existing); });
  html += `</div>
    <div class="form-actions">
      <button class="btn btn-primary" onclick="submitForm('${kind}')">Guardar</button>
      <button class="btn btn-secondary" onclick="closeForm()">Cancelar</button>
    </div>`;

  document.getElementById('form-panel').innerHTML = html;
  document.getElementById('form-panel').style.display = 'block';
  document.getElementById('overlay').style.display = 'block';
}

function renderField(f, existing) {
  let val = existing ? (existing[f.name] ?? '') : '';
  if (!existing && f.defaultToday && !val) val = hoyISO();
  const id = 'f_' + f.name;

  if (f.type === 'select') {
    const options = DATA[f.source].map(item =>
      `<option value="${esc(item[f.optionValue])}" ${String(item[f.optionValue])===String(val)?'selected':''}>${esc(item[f.optionLabel])}</option>`
    ).join('');
    return `<div class="field"><label>${f.label}</label><select id="${id}"><option value="">Seleccioná...</option>${options}</select></div>`;
  }
  if (f.type === 'select-fixed') {
    const options = f.options.map(o => `<option value="${o}" ${o===val?'selected':''}>${o}</option>`).join('');
    return `<div class="field"><label>${f.label}</label><select id="${id}">${options}</select></div>`;
  }
  if (f.type === 'textarea') {
    return `<div class="field"><label>${f.label}</label><textarea id="${id}">${esc(val)}</textarea></div>`;
  }
  if (f.type === 'date') {
    return `<div class="field"><label>${f.label}</label><input type="date" id="${id}" value="${esc(val)}"></div>`;
  }
  if (f.type === 'number') {
    return `<div class="field"><label>${f.label}</label><input type="number" id="${id}" value="${esc(val)}"></div>`;
  }
  if (f.type === 'youtube') {
    return `<div class="field"><label>${f.label}</label><input type="text" id="${id}" value="${esc(val)}" placeholder="Pegá el link o el ID del video">
      <div class="field-hint">Podés pegar el link completo de YouTube; se guarda el ID automáticamente.</div></div>`;
  }
  if (f.type === 'image' || f.type === 'file') {
    const isImg = f.type === 'image';
    return `<div class="field"><label>${f.label}</label>
      <input type="hidden" id="${id}" value="${esc(val)}">
      <div class="upload-row">
        <input type="file" id="${id}_file" accept="${isImg ? 'image/*' : 'application/pdf'}" onchange="handleUpload('${id}', this, ${isImg})">
        ${isImg ? `<img class="upload-preview" id="${id}_preview" src="${esc(val)}" style="${val?'display:inline-block':''}">` : ''}
      </div>
      <div class="upload-status" id="${id}_status">${val ? 'Archivo cargado.' : ''}</div>
    </div>`;
  }
  const lock = (f.lockOnEdit && existing) ? 'disabled' : '';
  return `<div class="field"><label>${f.label}</label><input type="text" id="${id}" value="${esc(val)}" ${lock}></div>`;
}

async function guardarContenido() {
  const tipo = document.getElementById('cf_Tipo').value;
  const nombre = document.getElementById('cf_Nombre').value.trim();
  const orden = document.getElementById('cf_Orden').value;
  let link = document.getElementById('cf_Link').value.trim();

  if (!nombre) { showToast('Falta el nombre del contenido', true); return; }
  if (!link) { showToast('Falta el ' + (tipo === 'Video' ? 'link de YouTube' : 'archivo PDF'), true); return; }
  if (tipo === 'Video') link = extractYoutubeId(link);

  const data = { ID_Clase: contenidoClaseActual, Tipo: tipo, Nombre: nombre, Orden: orden, Link: link };

  try {
    if (contenidoEditando) {

    await loadAll();
    renderPanelContenidos();

  }
}

function fileToBase64(file) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = () => resolve(reader.result.split(',')[1]);
    reader.onerror = reject;
    reader.readAsDataURL(file);
  });
}

async function handleUpload(fieldId, input, isImage) {
  const file = input.files[0];
  if (!file) return;
  const statusEl = document.getElementById(fieldId + '_status');
  statusEl.textContent = 'Subiendo...';
  try {
    const base64Data = await fileToBase64(file);
    const result = await api('uploadFile', { fileName: file.name, mimeType: file.type, fileData: base64Data });
    if (result.success) {
      document.getElementById(fieldId).value = result.url;
      statusEl.textContent = 'Archivo cargado correctamente.';
      if (isImage) {
        const prev = document.getElementById(fieldId + '_preview');
        prev.src = result.url;
        prev.style.display = 'inline-block';
      }
    } else {
      statusEl.textContent = 'Error al subir: ' + (result.error || 'desconocido');
    }
  } catch (e) {
    statusEl.textContent = 'Error al subir: ' + e.message;
  }
}

function extractYoutubeId(value) {
  if (!value) return value;
  const patterns = [/[?&]v=([^&]+)/, /youtu\.be\/([^?&]+)/, /youtube\.com\/embed\/([^?&]+)/];
  for (const p of patterns) { const m = value.match(p); if (m) return m[1]; }
  return value;
}

async function submitForm(kind) {
  const def = FORM_DEFS[kind];
  const data = {};
  for (const f of def.fields) {
    const el = document.getElementById('f_' + f.name);
    if (!el) continue;
    let value = el.value;
    if (f.type === 'youtube') value = extractYoutubeId(value);
    if (f.required && !value) {
      showToast('Falta completar: ' + f.label, true);
      return;
    }
    data[f.name] = value;
  }

  try {
    if (editingRow) {
      await api(def.updAction, { row: editingRow, data });
      showToast(def.title + ' actualizado correctamente.');
    } else {
      await api(def.addAction, { data });
      showToast(def.title + ' agregado correctamente.');
    }
    closeForm();
    await loadAll();
  } catch (e) {
    showToast('Error al guardar: ' + e.message, true);
  }
}

let toastTimer;
function showToast(msg, isError) {
  const el = document.getElementById('toast');
  el.textContent = msg;
  el.className = 'toast show' + (isError ? ' error' : '');
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => { el.className = 'toast' + (isError ? ' error' : ''); }, 3500);
}

function onAlumnoEgresadoChange() {
  const dni = document.getElementById('ef_Alumno').value;
  const inputDNI = document.getElementById('ef_DNI');
  const selectCurso = document.getElementById('ef_Curso');
  const hint = document.getElementById('ef_CursoHint');

  inputDNI.value = '';
  selectCurso.innerHTML = '';
  hint.textContent = '';

  if (!dni) {
    selectCurso.disabled = true;
    selectCurso.innerHTML = '<option value="">Primero elegí un alumno</option>';
    return;
  }

  inputDNI.value = dni;

  const alumno = DATA.alumnosConCursos.find(a => String(a.ID_Alumno) === String(dni));
  if (!alumno || !alumno.Cursos || alumno.Cursos.length === 0) {
    selectCurso.disabled = true;
    selectCurso.innerHTML = '<option value="">Sin cursos habilitados</option>';
    hint.textContent = 'Este alumno no tiene cursos habilitados. Verificá las inscripciones.';
    hint.style.color = 'var(--error)';
    return;
  }

  selectCurso.disabled = false;
  selectCurso.innerHTML = '<option value="">Seleccioná un curso...</option>';
  alumno.Cursos.forEach(c => {
    const opt = document.createElement('option');
    opt.value = c.Nombre;
    opt.textContent = c.Nombre;
    selectCurso.appendChild(opt);
  });
  hint.textContent = alumno.Cursos.length + ' curso(s) disponible(s) para este alumno.';
  hint.style.color = 'var(--ink-soft)';
}

async function submitEgresado() {
  const dniRaw = document.getElementById('ef_Alumno').value;
  const curso = document.getElementById('ef_Curso').value;
  const tipo = document.getElementById('ef_Tipo').value;
  const fecha = document.getElementById('ef_Fecha').value.trim();
  const acta = document.getElementById('ef_Acta').value.trim();
  const certificado = document.getElementById('ef_Certificado').value.trim();
  const estado = document.getElementById('ef_Estado').value;

  const dni = String(dniRaw || '').replace(/\./g, '').trim();

  if (!dni) { showToast('Falta elegir un alumno.', true); return; }
  if (!curso) { showToast('Falta elegir un curso.', true); return; }
  if (!fecha) { showToast('Falta la fecha.', true); return; }
  if (!acta) { showToast('Falta el acta.', true); return; }
  if (!certificado) { showToast('Falta el certificado.', true); return; }

  const alumno = DATA.alumnosConCursos.find(a => String(a.ID_Alumno).replace(/\./g,'') === dni);
  const nombreAlumno = alumno ? alumno.Nombre : '';

  const data = {
    Fecha: fecha,
    Nombre_Alumno: nombreAlumno,
    Curso: curso,
    DNI: dni,
    Acta: acta,
    Certificado: certificado,
    Tipo: tipo,
    Estado: estado
  };

  try {
    const res = await api('addEgresado', { data });
    if (res.error) throw new Error(res.error);
    showToast('Egresado agregado correctamente.');
    closeForm();
    await loadAll();
  } catch (e) {
    showToast('Error al guardar: ' + e.message, true);
  }
}
